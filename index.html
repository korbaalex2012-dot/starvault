// GAME STATE let userBalance = 500; let currentMode = "logic"; // "logic" или "speed" let currentCase = null; let secretCode = ""; let remainingAttempts = 0; let remainingTime = 0; let timerInterval = null; // SELECT MODE function selectMode(mode) { currentMode = mode; document.querySelectorAll('.mode-btn').forEach(btn => { btn.classList.toggle('active', btn.dataset.mode === mode); }); } // START GAME function startGame(caseKey) { const caseData = CASE_TYPES[caseKey]; if (userBalance < caseData.cost) { alert("Недостаточно монет для взлома этого сейфа!"); return; } // Списание стоимости userBalance -= caseData.cost; updateBalanceUI(); currentCase = caseData; secretCode = generateCode(caseData.digits); // Очистка истории и инпута document.getElementById('attempts-history').innerHTML = ""; document.getElementById('code-input').value = ""; document.getElementById('code-input').maxLength = caseData.digits; // Настройка UI игрового экрана document.getElementById('mode-selector').classList.add('hidden'); document.getElementById('case-selector').classList.add('hidden'); document.getElementById('game-screen').classList.remove('hidden'); document.getElementById('current-case-title').innerText = caseData.name; // Запуск выбранного режима if (currentMode === "logic") { remainingAttempts = caseData.logicAttempts; updateStatusUI(`Осталось попыток: ${remainingAttempts}`); } else { remainingTime = caseData.speedTime; updateStatusUI(`⏱️ Осталось: ${remainingTime} сек.`); clearInterval(timerInterval); timerInterval = setInterval(() => { remainingTime--; updateStatusUI(`⏱️ Осталось: ${remainingTime} сек.`); if (remainingTime <= 0) { clearInterval(timerInterval); endGame(false, "Время вышло! Сейф заблокировался."); } }, 1000); } } // MAKE ATTEMPT function makeAttempt() { const inputEl = document.getElementById('code-input'); const userGuess = inputEl.value.trim(); if (userGuess.length !== currentCase.digits) { alert(`Код должен состоять из ${currentCase.digits} цифр!`); return; } // Проверка кода const checkResult = checkCode(secretCode, userGuess); renderAttempt(checkResult); inputEl.value = ""; // Проверка на ПOБЕДУ const isWin = checkResult.every(item => item.status === "correct"); if (isWin) { if (timerInterval) clearInterval(timerInterval); userBalance += currentCase.reward; updateBalanceUI(); endGame(true, ` СЕЙФ ВЗЛОМАН! Вы забрали ${currentCase.reward} 🪙`); return; } // Если режим «Логика», уменьшаем попытки if (currentMode === "logic") { remainingAttempts--; updateStatusUI(`Осталось попыток: ${remainingAttempts}`); if (remainingAttempts <= 0) { endGame(false, `Попытки закончились! Код был: ${secretCode}`); } } } // WORDLE ALGORITHM function checkCode(secret, guess) { const result = []; const secretArr = secret.split(''); const guessArr = guess.split(''); // 1. Зелёные (на своих местах) for (let i = 0; i < guessArr.length; i++) { if (guessArr[i] === secretArr[i]) { result[i] = { digit: guessArr[i], status: "correct" }; secretArr[i] = null; } else { result[i] = null; } } // 2. Жёлтые (не на своих местах) и Серые (нет) for (let i = 0; i < guessArr.length; i++) { if (result[i] !== null) continue; const indexInSecret = secretArr.indexOf(guessArr[i]); if (indexInSecret !== -1) { result[i] = { digit: guessArr[i], status: "wrong-place" }; secretArr[indexInSecret] = null; } else { result[i] = { digit: guessArr[i], status: "absent" }; } } return result; } // RENDER ATTEMPT ROW function renderAttempt(result) { const historyContainer = document.getElementById('attempts-history'); const row = document.createElement('div'); row.className = 'attempt-row'; result.forEach(item => { const box = document.createElement('div'); box.className = `digit-box ${item.status}`; box.innerText = item.digit; row.appendChild(box); }); historyContainer.appendChild(row); historyContainer.scrollTop = historyContainer.scrollHeight; } // END GAME function endGame(isWin, message) { alert(message); quitGame(); } // QUIT / RESET function quitGame() { if (timerInterval) clearInterval(timerInterval); document.getElementById('game-screen').classList.add('hidden'); document.getElementById('mode-selector').classList.remove('hidden'); document.getElementById('case-selector').classList.remove('hidden'); } // HELPERS function generateCode(length) { let code = ""; for (let i = 0; i < length; i++) { code += Math.floor(Math.random() * 10).toString(); } return code; } function updateBalanceUI() { document.getElementById('user-balance').innerText = userBalance; } function updateStatusUI(text) { document.getElementById('status-text').innerText = text; } // Обработка Enter и авто-запуск document.getElementById('code-input').addEventListener('keypress', function (e) { if (e.key === 'Enter') { makeAttempt(); } }); document.addEventListener('DOMContentLoaded', () => { updateBalanceUI(); });
